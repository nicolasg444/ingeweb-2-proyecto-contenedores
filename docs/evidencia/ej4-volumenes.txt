===== Hallazgo extra: ¿dónde quedaron los datos de db-efimera? =====
# La imagen postgres declara VOLUME /var/lib/postgresql/data. Sin -v, Docker crea un volumen
# ANÓNIMO por contenedor. 'docker rm -f' (sin -v) no lo borra: queda huérfano con los datos.
$ docker volume ls -f dangling=true
DRIVER    VOLUME NAME
local     95757ae1a1bf27d77a03c2bb04fead538ec8d216697624c8f1b72e58fbd71ecb

$ docker run -d --name forense -v 95757ae1a1bf27d77a03c2bb04fead538ec8d216697624c8f1b72e58fbd71ecb:/var/lib/postgresql/data postgres:17-alpine
$ docker exec forense psql -U mercado_app -d mercado -c 'SELECT * FROM prueba;'
 n 
---
 1
(1 row)

# => el dato seguía ahí, pero nadie sabe que existe: un volumen anónimo no sirve para persistir.
#    Además nada lo borra solo: ocupa disco hasta 'docker volume prune' o 'docker rm -v'.

$ docker rm -f forense && docker volume rm 95757ae1a1bf27d77a03c2bb04fead538ec8d216697624c8f1b72e58fbd71ecb
forense

===== Paso 4: bind mount de solo lectura para los scripts de inicialización =====
# Adaptado al reto: superusuario mercado_admin + rol de aplicación mercado_app (02_rol_app.sh)
$ docker run -d --name db-init -e POSTGRES_DB=mercado -e POSTGRES_USER=mercado_admin \
    -e POSTGRES_PASSWORD_FILE=/run/secrets/db_password -e APP_DB_USER=mercado_app \
    -e APP_DB_PASSWORD_FILE=/run/secrets/db_app_password \
    -v "$(pwd)/secrets/db_password.txt:/run/secrets/db_password:ro" \
    -v "$(pwd)/secrets/db_app_password.txt:/run/secrets/db_app_password:ro" \
    -v pgdata:/var/lib/postgresql/data -v "$(pwd)/db/init:/docker-entrypoint-initdb.d:ro" postgres:17-alpine
